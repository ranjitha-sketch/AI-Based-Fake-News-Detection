import os
import re
import string
import pandas as pd
import tkinter as tk

from tkinter import messagebox, scrolledtext, filedialog
from datetime import datetime

from sklearn.model_selection import train_test_split
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.linear_model import LogisticRegression
from sklearn.pipeline import Pipeline
from sklearn.metrics import accuracy_score, classification_report


# ======================== CONFIGURATION ========================

DATASET_FILE = "news.csv"
HISTORY_FILE = "prediction_history.csv"

LABELS = {
    0: "REAL NEWS",
    1: "FAKE NEWS"
}

model = None
model_accuracy = 0.0


# ======================== TEXT PREPROCESSING ========================

def clean_text(text):
    """Clean news text using basic NLP preprocessing."""

    text = str(text).lower()
    text = re.sub(r"https?://\S+|www\.\S+", "", text)
    text = re.sub(r"\S+@\S+", "", text)
    text = re.sub(r"<.*?>", "", text)
    text = re.sub(r"\d+", "", text)
    text = text.translate(str.maketrans("", "", string.punctuation))
    text = re.sub(r"\s+", " ", text).strip()

    return text


# ======================== DATASET MANAGEMENT ========================

def load_dataset():
    """Load and validate the training dataset."""

    if not os.path.exists(DATASET_FILE):
        messagebox.showerror(
            "Dataset Missing",
            "The file news.csv was not found.\n\n"
            "Create a CSV file with these columns:\n"
            "text,label\n\n"
            "Label 0 = Real news\n"
            "Label 1 = Fake news"
        )
        return None

    try:
        data = pd.read_csv(DATASET_FILE)

        required_columns = {"text", "label"}

        if not required_columns.issubset(data.columns):
            messagebox.showerror(
                "Invalid Dataset",
                "The dataset must contain 'text' and 'label' columns."
            )
            return None

        data = data.dropna(subset=["text", "label"]).copy()
        data["text"] = data["text"].apply(clean_text)
        data["label"] = pd.to_numeric(data["label"], errors="coerce")
        data = data.dropna(subset=["label"])

        data = data[data["label"].isin([0, 1])]
        data = data[data["text"].str.len() > 0]
        data["label"] = data["label"].astype(int)

        if data["label"].nunique() != 2:
            messagebox.showerror(
                "Dataset Error",
                "The dataset must contain both real and fake news."
            )
            return None

        if data["label"].value_counts().min() < 2:
            messagebox.showerror(
                "Dataset Error",
                "At least two examples of each class are required."
            )
            return None

        return data

    except Exception as error:
        messagebox.showerror("Dataset Error", str(error))
        return None


# ======================== MODEL TRAINING ========================

def train_model():
    """Train and evaluate the fake news detection model."""

    global model, model_accuracy

    data = load_dataset()

    if data is None:
        return

    try:
        x_train, x_test, y_train, y_test = train_test_split(
            data["text"],
            data["label"],
            test_size=0.20,
            random_state=42,
            stratify=data["label"]
        )

        model = Pipeline([
            (
                "tfidf",
                TfidfVectorizer(
                    stop_words="english",
                    max_features=20000,
                    ngram_range=(1, 2),
                    sublinear_tf=True
                )
            ),
            (
                "classifier",
                LogisticRegression(
                    max_iter=1000,
                    class_weight="balanced",
                    random_state=42
                )
            )
        ])

        model.fit(x_train, y_train)

        predictions = model.predict(x_test)
        model_accuracy = accuracy_score(y_test, predictions) * 100

        report = classification_report(
            y_test,
            predictions,
            labels=[0, 1],
            target_names=["REAL NEWS", "FAKE NEWS"],
            zero_division=0
        )

        accuracy_label.config(
            text=f"Model Test Accuracy: {model_accuracy:.2f}%"
        )

        dataset_label.config(
            text=f"Training Dataset: {len(data)} articles"
        )

        result_area.delete("1.0", tk.END)
        result_area.insert(
            tk.END,
            "MODEL TRAINING COMPLETED SUCCESSFULLY\n\n"
            f"Total articles: {len(data)}\n"
            f"Training articles: {len(x_train)}\n"
            f"Testing articles: {len(x_test)}\n"
            f"Test accuracy: {model_accuracy:.2f}%\n\n"
            "CLASSIFICATION REPORT\n"
            "-----------------------------\n"
            f"{report}\n"
            "Note: Test accuracy does not guarantee "
            "correct predictions on new articles."
        )

        messagebox.showinfo(
            "Training Complete",
            "The AI model has been trained successfully."
        )

    except Exception as error:
        model = None
        messagebox.showerror(
            "Training Error",
            f"Unable to train the model:\n{error}"
        )


# ======================== PREDICTION HISTORY ========================

def save_history(news, prediction, confidence):
    """Save predictions locally in a CSV file."""

    record = {
        "date_time": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
        "news_text": news,
        "prediction": prediction,
        "confidence_percent": round(confidence, 2)
    }

    try:
        file_exists = os.path.exists(HISTORY_FILE)

        pd.DataFrame([record]).to_csv(
            HISTORY_FILE,
            mode="a",
            header=not file_exists,
            index=False,
            encoding="utf-8"
        )

    except Exception as error:
        messagebox.showwarning(
            "History Warning",
            f"Prediction completed, but history was not saved:\n{error}"
        )


# ======================== FAKE NEWS DETECTION ========================

def detect_news():
    """Classify the entered article using the trained model."""

    if model is None:
        messagebox.showwarning(
            "Model Not Trained",
            "Please train the AI model before detecting news."
        )
        return

    news = news_input.get("1.0", tk.END).strip()

    if len(news) < 10:
        messagebox.showwarning(
            "Input Required",
            "Please enter a news headline or article."
        )
        return

    try:
        cleaned_news = clean_text(news)

        if not cleaned_news:
            messagebox.showwarning(
                "Invalid Input",
                "Please enter meaningful news text."
            )
            return

        prediction = int(model.predict([cleaned_news])[0])
        probabilities = model.predict_proba([cleaned_news])[0]

        class_positions = list(model.classes_)
        prediction_index = class_positions.index(prediction)
        confidence = float(probabilities[prediction_index] * 100)

        label = LABELS[prediction]

        result_area.delete("1.0", tk.END)
        result_area.insert(
            tk.END,
            "NEWS DETECTION RESULT\n"
            "================================\n\n"
            f"Prediction: {label}\n"
            f"Model confidence: {confidence:.2f}%\n\n"
            f"Real news score: "
            f"{probabilities[class_positions.index(0)] * 100:.2f}%\n"
            f"Fake news score: "
            f"{probabilities[class_positions.index(1)] * 100:.2f}%\n\n"
            "IMPORTANT INFORMATION\n"
            "--------------------------------\n"
            "This result is based on patterns learned from "
            "the training dataset. It does not independently "
            "verify facts or confirm whether the story is true.\n\n"
            "Check the original source, publication date, "
            "supporting evidence, and trusted news organizations."
        )

        save_history(news, label, confidence)

    except Exception as error:
        messagebox.showerror(
            "Prediction Error",
            f"Unable to classify this news:\n{error}"
        )

        return

        history = pd.read_csv(HISTORY_FILE)

        history_window.geometry("850x500")
        history_window.configure(bg="#f4f6fb")

        heading = tk.Label(
            history_window,
            text="PREDICTION HISTORY",
            font=("Arial", 16, "bold"),
            bg="#f4f6fb",
            fg="#243b67"
        )
        heading.pack(pady=10)

        history_text = scrolledtext.ScrolledText(
            history_window,
            wrap=tk.WORD,
            font=("Consolas", 10),
            width=100,
            height=25
        )
        history_text.pack(padx=10, pady=10, fill=tk.BOTH, expand=True)

        history_text.insert(tk.END, history.to_string(index=False))
        history_text.config(state=tk.DISABLED)

    except Exception as error:
        messagebox.showerror("History Error", str(error))


# ======================== EXPORT HISTORY ========================

def export_history():
    """Allow the user to export prediction history."""

    if not os.path.exists(HISTORY_FILE):
        messagebox.showinfo(
            "No History",
            "There is no prediction history to export."
        )
        return

    destination = filedialog.asksaveasfilename(
        title="Export Prediction History",
        defaultextension=".csv",
        filetypes=[("CSV files", "*.csv")]
    )

    if not destination:
        return

    try:
        history = pd.read_csv(HISTORY_FILE)
        history.to_csv(destination, index=False, encoding="utf-8")

        messagebox.showinfo(
            "Export Successful",
            f"History exported to:\n{destination}"
        )

    except Exception as error:
        messagebox.showerror("Export Error", str(error))
